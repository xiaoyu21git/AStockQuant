```python
from dataclasses import dataclass, field
from enum import Enum
from typing import List, Dict, Optional

class FinancialEventType(str, Enum):
    """金融事件类型 (与 C++ EventTypes::NEWS_* 一一对应)"""
    EARNINGS      = "news.earnings"       # 业绩预告/快报
    POLICY        = "news.policy"         # 政策/监管
    MATERIAL      = "news.material"       # 重大事项(重组/增减持/分红)
    SOCIAL        = "news.social"         # 社交媒体情绪异动
    QUOTE_ALERT   = "news.quote_alert"    # 行情异动(放量/涨跌停/盘口)
    ALERT         = "news.alert"          # 突发/快讯

class InfoSource(str, Enum):
    """信息源标识"""
    EASTMONEY   = "eastmoney"     # 东方财富
    CNINFO      = "cninfo"        # 巨潮资讯
    XUEQIU      = "xueqiu"        # 雪球
    CLS         = "cls"           # 财联社
    SINA        = "sina"          # 新浪财经
    GM_SDK      = "gm_sdk"        # 掘金SDK (行情异动)

@dataclass
class FinancialEvent:
    """标准化金融事件对象"""
    source: str                              # 信息源标识
    title: str                               # 标题
    summary: str                             # 摘要 (≤500字)
    symbols: List[str] = field(default_factory=list)  # 关联标的代码 ["000001","600000"]
    sentiment_score: float = 0.0             # 情感分 -1.0 ~ 1.0
    event_type: FinancialEventType = FinancialEventType.ALERT
    confidence: float = 0.0                  # NLP 置信度 0.0 ~ 1.0
    tags: Dict[str, str] = field(default_factory=dict)  # {"超预期":"true","风险类型":"立案调查"}
    timestamp: int = 0                       # Unix 毫秒

    def to_event_format_data(self) -> dict:
        """转换为 C++ EventFormat.data + metadata"""
        return {
            "data": {
                "source":          self.source,
                "title":           self.title,
                "summary":         self.summary,
                "symbols":         self.symbols,
                "sentiment_score": str(self.sentiment_score),
                "confidence":      str(self.confidence),
            },
            "metadata": {
                "timestamp":   str(self.timestamp),
                **{f"tag.{k}": v for k, v in self.tags.items()},
            }
        }
```

### 2.3 financial_lexicon.py — 金融情感词典

```python
"""
金融情感词典模块
- 加载 JSON 词典文件
- 提供词语级/词组级情感值查询
- 程度副词加权 + 否定词翻转
"""

import json
import os
from typing import Dict, Optional, Tuple

class FinancialLexicon:
    """金融情感词典引擎"""

    # 程度副词权重
    DEGREE_ADVERBS: Dict[str, float] = {
        "大幅": 1.5, "显著": 1.3, "明显": 1.2, "小幅": 0.7,
        "略微": 0.5, "微幅": 0.4, "极大": 1.6, "严重": 1.5,
    }

    # 否定词 (翻转后续情感)
    NEGATION_WORDS: set = {"不", "未", "无", "非", "没", "难以", "未能", "无法"}

    # 否定窗口大小 (否定词后多少词受影响)
    NEGATION_WINDOW: int = 4

    def __init__(self, lexicon_path: Optional[str] = None):
        if lexicon_path is None:
            lexicon_path = os.path.join(
                os.path.dirname(__file__),
                "financial_sentiment_lexicon.json"
            )
        with open(lexicon_path, "r", encoding="utf-8") as f:
            data = json.load(f)
        self._words: Dict[str, float] = data.get("words", {})
        self._phrases: Dict[str, float] = data.get("phrases", {})

    def word_score(self, word: str) -> float:
        """查询单个词的情感值"""
        return self._words.get(word, 0.0)

    def phrase_score(self, phrase: str) -> float:
        """查询词组的情感值 (优先长匹配)"""
        if phrase in self._phrases:
            return self._phrases[phrase]
        return self.word_score(phrase)

    def score_sequence(
        self, tokens: list[str]
    ) -> Tuple[float, list[Tuple[str, float]]]:
        """
        对分词序列计算加权情感值

        Args:
            tokens: HanLP 分词后的词组序列

        Returns:
            (final_score, trace): 最终情感分 + 每个词的得分追踪
        """
        total_score = 0.0
        trace = []
        negated = False
        neg_window = 0
        degree_mult = 1.0

        for i, token in enumerate(tokens):
            # 检查程度副词 (作用于下一个情感词)
            if token in self.DEGREE_ADVERBS:
                degree_mult = self.DEGREE_ADVERBS[token]
                continue

            # 检查否定词 (翻转后续 N 个词)
            if token in self.NEGATION_WORDS:
                negated = True
                neg_window = self.NEGATION_WINDOW
                degree_mult = 1.0
                continue

            # 否定窗口递减
            if negated:
                neg_window -= 1
                if neg_window <= 0:
                    negated = False

            # 查询情感值
            s = self.phrase_score(token)
            if s != 0.0:
                s = s * degree_mult
                if negated:
                    s = -s
                total_score += s
                trace.append((token, s))

            degree_mult = 1.0  # 重置程度副词

        # 归一化到 -1.0 ~ 1.0
        final = max(-1.0, min(1.0, total_score / max(1, len(tokens) * 0.3)))
        return final, trace


class SentimentAnalyzer:
    """金融情感分析器 = FinancialLexicon + HanLP 分词"""

    def __init__(self, lexicon: FinancialLexicon):
        self._lexicon = lexicon
        self._tokenizer = None  # 延迟加载 HanLP

    def _ensure_tokenizer(self):
        if self._tokenizer is None:
            import hanlp
            self._tokenizer = hanlp.load(
                hanlp.pretrained.tok.COARSE_ELECTRA_SMALL_ZH
            )

    def analyze(self, text: str) -> Tuple[float, dict]:
        """
        分析文本的金融情感

        Returns:
            (score, detail): 情感分 + 详细追踪信息
        """
        self._ensure_tokenizer()
        tokens = self._tokenizer(text)
        score, trace = self._lexicon.score_sequence(tokens)

        detail = {
            "tokens": tokens,
            "trace": [{"word": w, "score": s} for w, s in trace],
            "token_count": len(tokens),
            "matched_count": len(trace),
        }
        return score, detail
```

### 2.5 nlp_pipeline.py — NLP 分析流水线

```python
"""
NLP 分析流水线
- 输入: 原始文本 (title + content)
- 输出: FinancialEvent (含 sentiment, entities, type, tags)
"""

import time
from typing import List, Optional

from .event_types import FinancialEvent, FinancialEventType, InfoSource
from .financial_lexicon import FinancialLexicon, SentimentAnalyzer


class EntityRecognizer:
    """实体识别器: 从文本中提取关联标的"""

    def __init__(self, lexicon: FinancialLexicon):
        self._ner = None  # 延迟加载 HanLP NER
        # 标的名称/代码映射表 (从 MarketDataService 或 Redis 加载)
        self._stock_names: dict[str, str] = {}   # 公司简称 → 代码
        self._stock_codes: set[str] = set()      # 所有有效代码

    def _ensure_ner(self):
        if self._ner is None:
            import hanlp
            self._ner = hanlp.load(
                hanlp.pretrained.ner.MSRA_NER_ELECTRA_SMALL_ZH
            )

    def load_stock_registry(self, stocks: list[tuple[str, str, str]]):
        """
        加载标的注册表
        Args:
            stocks: [(代码, 简称, 全称), ...]
                    例: [("000001", "平安银行", "平安银行股份有限公司"), ...]
        """
        for code, short_name, full_name in stocks:
            self._stock_names[short_name] = code
            self._stock_names[full_name] = code
            # 处理去后缀的简称 ("平安银行" 和 "平安银行股份" 都能匹配)
            if "股份" in full_name:
                alias = full_name.replace("股份有限公司", "").replace("有限公司", "")
                if alias != short_name:
                    self._stock_names[alias] = code
            self._stock_codes.add(code)

    def extract(self, text: str) -> List[str]:
        """从文本中提取关联标的代码列表"""
        self._ensure_ner()
        # HanLP NER 提取实体
        result = self._ner(text)
        entities = []
        for entity_info in result:
            entity_text = entity_info[0]   # 实体文本
            entity_type = entity_info[1]   # 实体类型: PERSON/ORGANIZATION/LOCATION

            if entity_type == "ORGANIZATION":
                # 尝试匹配公司名称
                code = self._stock_names.get(entity_text)
                if code:
                    entities.append(code)

        # 补充: 正则匹配 6 位数字代码
        import re
        code_pattern = re.compile(r'\b(\d{6})\b')
        for m in code_pattern.finditer(text):
            code = m.group(1)
            if code in self._stock_codes and code not in entities:
                entities.append(code)

        return entities


class EventClassifier:
    """事件分类器: 将文本归类到 FinancialEventType"""

    # 关键词 → 事件类型的规则映射
    TYPE_RULES = [
        (FinancialEventType.EARNINGS,
         ["业绩预告", "业绩快报", "年报", "季报", "半年报", "净利润",
          "营收", "每股收益", "预增", "预减", "预亏", "扭亏"]),
        (FinancialEventType.POLICY,
         ["证监会", "银保监", "央行", "国务院", "发改委", "工信部",
          "产业政策", "监管政策", "行业规范", "管理办法"]),
        (FinancialEventType.MATERIAL,
         ["重组", "收购", "并购", "增发", "配股", "资产注入", "整体上市",
          "增持", "减持", "回购", "分红", "送转", "股权激励", "员工持股",
          "重大合同", "中标", "停牌", "复牌"]),
        (FinancialEventType.QUOTE_ALERT,
         ["涨停", "跌停", "放量", "缩量", "异动", "拉升", "跳水",
          "新高", "新低", "突破", "破位"]),
        (FinancialEventType.SOCIAL,
         ["股吧", "雪球", "热议", "热搜", "讨论", "关注"]),
    ]

    def classify(self, text: str,
                 source: InfoSource) -> FinancialEventType:
        """基于关键词匹配的事件分类"""
        scores = {t: 0 for t in FinancialEventType}
        for event_type, keywords in self.TYPE_RULES:
            for kw in keywords:
                if kw in text:
                    scores[event_type] += 1

        # 社交来源偏向 SOCIAL 类型
        if source in (InfoSource.XUEQIU, InfoSource.CLS):
            scores[FinancialEventType.SOCIAL] += 1

        # 返回最高分类型, 默认 ALERT
        best = max(scores, key=scores.get)
        return best if scores[best] > 0 else FinancialEventType.ALERT


class TagExtractor:
    """标签提取器: 从文本中提取结构化标签"""

    TAG_PATTERNS = {
        "超预期": ["超预期", "好于预期", "优于预期", "超出市场预期"],
        "低于预期": ["低于预期", "不及预期", "差于预期", "不达预期"],
        "立案调查": ["立案调查", "被证监会调查", "涉嫌信息披露违规"],
        "问询函": ["问询函", "关注函", "监管函", "警示函"],
        "ST警示": ["ST", "*ST", "退市风险警示", "实施其他风险警示"],
        "高分红": ["高分红", "高送转", "高比例分红", "大比例分红"],
        "大额减持": ["大额减持", "清仓减持", "大幅减持", "减持计划"],
        "回购计划": ["回购计划", "股份回购", "回购方案", "回购注销"],
    }

    def extract(self, text: str) -> dict[str, str]:
        """提取标签 → {"超预期": "true", "问询函": "true", ...}"""
        tags = {}
        for tag_name, patterns in self.TAG_PATTERNS.items():
            for pat in patterns:
                if pat in text:
                    tags[tag_name] = "true"
                    break
        return tags


class NLPPipeline:
    """NLP 分析流水线: 实体识别 + 情感分析 + 事件分类 + 标签提取"""

    def __init__(
        self,
        lexicon: FinancialLexicon,
        entity_recognizer: EntityRecognizer,
        classifier: EventClassifier,
        tag_extractor: TagExtractor,
    ):
        self._sentiment = SentimentAnalyzer(lexicon)
        self._entities = entity_recognizer
        self._classifier = classifier
        self._tags = tag_extractor

    def process(
        self,
        title: str,
        content: str,
        source: InfoSource,
    ) -> FinancialEvent:
        """
        完整 NLP 分析流水线

        Args:
            title: 新闻标题
            content: 新闻正文
            source: 信息来源

        Returns:
            完整标注的 FinancialEvent
        """
        # 合并文本用于分析
        full_text = f"{title}。{content[:500]}"  # 标题+正文前500字
        summary = content[:200] if len(content) > 200 else content

        # 1. 实体识别 (提取关联标的)
        symbols = self._entities.extract(full_text)

        # 2. 情感分析
        score, detail = self._sentiment.analyze(full_text)
        confidence = min(1.0, detail["matched_count"] / max(1, detail["token_count"]))

        # 3. 事件分类
        event_type = self._classifier.classify(full_text, source)

        # 4. 标签提取
        tags = self._tags.extract(full_text)

        return FinancialEvent(
            source=source.value,
            title=title,
            summary=summary,
            symbols=symbols,
            sentiment_score=score,
            event_type=event_type,
            confidence=confidence,
            tags=tags,
            timestamp=int(time.time() * 1000),
        )
```

### 2.6 adapter.py — 多源信息适配器

```python
"""
多源信息适配器
- 每个信息源一个 Adapter 子类
- 负责: 拉取/接收 → 去重 → 清洗 → 送入 NLP Pipeline
"""

import asyncio
import hashlib
import logging
from abc import ABC, abstractmethod
from typing import List, AsyncIterator, Optional

import aiohttp

from .event_types import FinancialEvent, InfoSource
from .nlp_pipeline import NLPPipeline
from .publisher import EventPublisher


class BaseAdapter(ABC):
    """信息源适配器基类"""

    def __init__(self, source: InfoSource, pipeline: NLPPipeline,
                 publisher: EventPublisher, redis_url: str = ""):
        self._source = source
        self._pipeline = pipeline
        self._publisher = publisher
        self._seen_fingerprints: set[str] = set()  # 内存去重
        self._redis = None  # Redis 持久化去重 (可选)

    def _fingerprint(self, title: str, content: str) -> str:
        """URL/内容指纹去重"""
        h = hashlib.sha256(f"{title}|{content[:200]}".encode())
        return h.hexdigest()[:16]

    def _is_duplicate(self, fp: str) -> bool:
        if fp in self._seen_fingerprints:
            return True
        self._seen_fingerprints.add(fp)
        # TODO: 超过 10 万条时清理旧指纹
        return False

    def _process_item(self, title: str, content: str) -> Optional[FinancialEvent]:
        """处理单条信息"""
        fp = self._fingerprint(title, content)
        if self._is_duplicate(fp):
            logging.debug(f"[{self._source.value}] 重复跳过: {title[:40]}")
            return None

        try:
            event = self._pipeline.process(title, content, self._source)
            if not event.symbols:
                logging.debug(f"[{self._source.value}] 无关联标的: {title[:40]}")
                return None  # 不关联任何标的的新闻丢弃
            return event
        except Exception as e:
            logging.error(f"[{self._source.value}] NLP异常: {title[:40]} | {e}")
            return None

    @abstractmethod
    async def poll(self) -> List[FinancialEvent]:
        """拉取一批新闻并返回金融事件列表"""
        ...


class AKShareNewsAdapter(BaseAdapter):
    """AKShare 新闻适配器 (东方财富/巨潮资讯/雪球)"""

    def __init__(self, pipeline: NLPPipeline, publisher: EventPublisher):
        super().__init__(InfoSource.EASTMONEY, pipeline, publisher)
        self._session: Optional[aiohttp.ClientSession] = None

    async def _ensure_session(self):
        if self._session is None:
            self._session = aiohttp.ClientSession()

    async def _fetch_eastmoney_news(self) -> list[dict]:
        """东方财富 - 7x24 快讯"""
        import akshare as ak
        try:
            df = ak.stock_zh_a_alerts_cls()
            if df is None or df.empty:
                return []
            return df.to_dict("records")
        except Exception as e:
```

### 2.7 publisher.py — 事件发布客户端

```python
"""
金融事件发布客户端
将 FinancialEvent 转化为 C++ EventFormat 并发布到 EventBus
"""

import logging
from typing import Optional

from .event_types import FinancialEvent


class EventPublisher:
    """金融事件 → C++ EventBus 发布器"""

    def __init__(self):
        self._bus = None  # 延迟获取 EventBus

    def _ensure_bus(self):
        """延迟获取全局 EventBus (避免循环导入)"""
        if self._bus is not None:
            return self._bus
        try:
            from astock_engine.core.eventbus_bridge_cpp_python import (
                get_engine_bus
            )
            self._bus = get_engine_bus()
        except ImportError:
            # 测试环境: 使用 Python EventBus fallback
            from astock_engine.core.eventbus_unified import UnifiedEventBus
            self._bus = UnifiedEventBus.instance()
        return self._bus

    def publish(self, event: FinancialEvent) -> bool:
        """
        发布金融事件到 C++ EventBus

        Args:
            event: 完整的 FinancialEvent

        Returns:
            True 发送成功
        """
        bus = self._ensure_bus()
        if bus is None:
            logging.warning("[EventPublisher] EventBus 不可用, 事件丢弃")
            return False

        event_format = self._build_event_format(event)

        try:
            bus.publish(event_format, priority=1)  # NORMAL priority
            logging.info(
                f"[EventPublisher] 发布 {event.event_type.value}: "
                f"{event.title[:50]} → {event.symbols} "
                f"sentiment={event.sentiment_score:.2f}"
            )
            return True
        except Exception as e:
            logging.error(f"[EventPublisher] 发布失败: {e}")
            return False

    def publish_batch(self, events: list[FinancialEvent]) -> int:
        """批量发布, 返回成功数"""
        count = 0
        for event in events:
            if self.publish(event):
                count += 1
        return count

    def _build_event_format(self, event: FinancialEvent):
        """将 FinancialEvent 转换为 EventFormat"""
        try:
            from eventbus_native import EventFormat, EventPriority

            fmt = EventFormat()
            fmt.type = event.event_type.value
            fmt.priority = EventPriority.NORMAL

            payload = event.to_event_format_data()
            for key, value in payload["data"].items():
                fmt.data[key] = value
            for key, value in payload["metadata"].items():
                fmt.metadata[key] = value

            return fmt
        except ImportError:
            # 测试环境 fallback
            return {
                "type": event.event_type.value,
                "data": event.to_event_format_data()["data"],
                "metadata": event.to_event_format_data()["metadata"],
            }
```

```cpp
/// @brief 事件缓存记录
struct EventRecord {
    std::string eventType;       // "news.earnings" 等
    double sentimentScore;       // -1.0 ~ 1.0
    double confidence;           // 0.0 ~ 1.0
    std::unordered_map<std::string, std::string> tags;

/// @brief 事件驱动因子
///
/// 数据来源: EventBus 上的 news.* 事件 (非传统行情/财务数据库)
/// 接入方式: 继承 BaseFactor, 通过 FactorInstanceManager 注册
/// 核心逻辑:
///   1. 在内存中维护每个标的的事件缓存 (通过 EventBus 订阅更新)
///   2. calculate() 时根据缓存中的事件情感分和标签计算因子值
///   3. 输出标准化: ZScore 或 Rank
class EventDrivenFactor final : public BaseFactor {
public:
    struct Params : CommonParams {
        double sentimentWeight = 0.2;     // 情感分权重
        double superExpectedBonus = 0.1;  // 超预期额外加分
        double investigationPenalty = 0.3; // 立案调查减分
        int maxEventAgeHours = 24;         // 事件有效期(小时)
        bool enableHanlpSentiment = true;  // 启用 HanLP 情感分
    };

    EventDrivenFactor() = default;
    ~EventDrivenFactor() override;

    // ── BaseFactor 接口 ──
    CalculationResult calculate(const CalculationContext& ctx) override;
    DataRequirements getDataRequirements() const override { return {}; }
    BoundaryRules getBoundaryRules() const override;
    int getLookbackDays() const override { return 1; }

    // ── EventBus 事件接收 (由 FactorInstanceManager 调用) ──
    void onEvent(const engine::EventFormat& event);

    // ── 配置 ──
    void loadConfig(const std::string& jsonConfig) override;
    static std::unique_ptr<BaseFactor> create(
        const FactorInstanceInfo& info,
        std::shared_ptr<DataAvailabilityChecker> checker);

private:
    Params m_params;
    mutable std::mutex m_cacheMutex;
    // symbol → 该标的关联的事件列表
    std::unordered_map<std::string, std::vector<EventRecord>> m_eventCache;

    double computeSignalScore(const std::string& symbol) const;
    void cleanExpiredEvents(int64_t nowMs);
```

```cpp
void EventDrivenFactor::onEvent(const engine::EventFormat& event) {
    std::lock_guard<std::mutex> lock(m_cacheMutex);

    // 解析事件中的标的列表
    const auto* symbolsPtr = event.getData<std::vector<std::string>>("symbols");
    if (!symbolsPtr || symbolsPtr->empty()) return;

    // 解析情感分
    double sentiment = 0.0;
    if (auto* s = event.getData<std::string>("sentiment_score")) {
        try { sentiment = std::stod(*s); } catch (...) {}
    }

    // 解析置信度
    double confidence = 0.0;
    if (auto* c = event.getData<std::string>("confidence")) {
        try { confidence = std::stod(*c); } catch (...) {}
    }

    // 解析标签
    std::unordered_map<std::string, std::string> tags;
    for (const auto& [key, val] : event.metadata()) {
        if (key.find("tag.") == 0) {
            tags[key.substr(4)] = val;
        }
    }

    // 解析时间戳
    int64_t ts = 0;
    if (auto* t = event.getMetadata("timestamp")) {
        try { ts = std::stoll(*t); } catch (...) {}
    }

    // 为每个关联标的添加事件记录
    EventRecord record{event.type(), sentiment, confidence, tags, ts};
    for (const auto& sym : *symbolsPtr) {
        m_eventCache[sym].push_back(record);
    }
}

// ═══════════════════════════════════════════════
// 因子计算
// ═══════════════════════════════════════════════

double EventDrivenFactor::computeSignalScore(const std::string& symbol) const {
    auto it = m_eventCache.find(symbol);
    if (it == m_eventCache.end() || it->second.empty()) {
        return 0.0;  // 无事件 → 中性
    }

    double totalScore = 0.0;
    int count = 0;

    for (const auto& record : it->second) {
        double eventScore = 0.0;

        // 1. 情感分贡献
        if (m_params.enableHanlpSentiment) {
            eventScore += record.sentimentScore * m_params.sentimentWeight;
        }

        // 2. 超预期标签
        auto tagsIt = record.tags.find("超预期");
        if (tagsIt != record.tags.end() && tagsIt->second == "true") {
            eventScore += m_params.superExpectedBonus;
        }

        // 3. 低于预期标签
        tagsIt = record.tags.find("低于预期");
        if (tagsIt != record.tags.end() && tagsIt->second == "true") {
            eventScore -= m_params.superExpectedBonus;
        }

        // 4. 立案调查 / ST 警示
        tagsIt = record.tags.find("立案调查");
        if (tagsIt != record.tags.end() && tagsIt->second == "true") {
            eventScore -= m_params.investigationPenalty;
        }
        tagsIt = record.tags.find("ST警示");
        if (tagsIt != record.tags.end() && tagsIt->second == "true") {
            eventScore -= m_params.investigationPenalty;
        }

        // 5. 置信度加权
        eventScore *= record.confidence;

        totalScore += eventScore;
        ++count;
    }

    return count > 0 ? totalScore / static_cast<double>(count) : 0.0;
}

void EventDrivenFactor::cleanExpiredEvents(int64_t nowMs) {
    int64_t cutoff = nowMs - m_params.maxEventAgeHours * 3600000LL;
    for (auto& [sym, records] : m_eventCache) {
        records.erase(
            std::remove_if(records.begin(), records.end(),
                [cutoff](const EventRecord& r) { return r.timestampMs < cutoff; }),
            records.end());
    }
    // 删除空的 symbol 条目
    for (auto it = m_eventCache.begin(); it != m_eventCache.end(); ) {
        if (it->second.empty())
            it = m_eventCache.erase(it);
        else
            ++it;
    }
}

CalculationResult EventDrivenFactor::calculate(const CalculationContext& ctx) {
    CalculationResult result;
    const auto& symbols = ctx.symbols;
    const auto& date = ctx.date;

    std::lock_guard<std::mutex> lock(m_cacheMutex);

    for (const auto& sym : symbols) {
        double score = computeSignalScore(sym);
        if (std::isfinite(score)) {
            result.values[sym] = score;
        }
    }

    // 应用标准化 (子类调用 BaseFactor 方法)
    if (m_params.standardization != StandardizationMethod::NONE) {
        applyCommonStandardization(
            result.values,
            m_params.standardization,
            m_params.neutralization,
            ctx);
    }

    return result;
}

BoundaryRules EventDrivenFactor::getBoundaryRules() const {
    BoundaryRules rules;
    rules.excludeNewStocks = false;     // 事件因子对次新股有效
    rules.excludeSuspended = true;      // 停牌股无成交, 排除
    rules.excludeDelisted = true;
    rules.outlierZScore = 5.0;          // 极端值截断
    return rules;
}
```

```cpp
void EventRiskSubscriber::onFinancialEvent(const engine::EventFormat& event) {
    // 解析标的列表
    const auto* symbolsPtr = event.getData<std::vector<std::string>>("symbols");
    if (!symbolsPtr) return;

    // 解析标签
    std::unordered_map<std::string, std::string> tags;
    for (const auto& [key, val] : event.metadata()) {
        if (key.find("tag.") == 0) {
            tags[key.substr(4)] = val;
        }
    }

    // 解析情感分和置信度
    double sentiment = 0.0;
    if (auto* s = event.getData<std::string>("sentiment_score")) {

    }
    double confidence = 0.0;
    if (auto* c = event.getData<std::string>("confidence")) {
        try { confidence = std::stod(*c); } catch (...) {}
    }

    // 对每个关联标的执行风控动作
    for (const auto& sym : *symbolsPtr) {
        applyEventTags(sym, tags, sentiment, confidence);
    }

void EventRiskSubscriber::applyEventTags(
    const std::string& symbol,
    const std::unordered_map<std::string, std::string>& tags,
    double sentiment, double confidence)
{
    bool configChanged = false;
    auto config = RiskManager::instance().riskConfig();  // 获取当前配置

    // ── 规则1: 立案调查 → 禁止开仓 + 收紧止损 ──
    auto it = tags.find("立案调查");
    if (it != tags.end() && it->second == "true") {
        m_blockedSymbols.insert(symbol);
        INTERNAL_WARN_STREAM << "[EventRisk] 立案调查封禁: " << symbol;

        // 保留原始配置
        if (m_overrides.find(symbol) == m_overrides.end()) {
            m_overrides[symbol] = {
                config.stopLossPercent,
                config.maxPositionPercent
            };
        }
        // 收紧止损: -10% → -5%
        config.stopLossPercent = std::min(config.stopLossPercent, 5.0);
        configChanged = true;
    }

    // ── 规则2: ST警示 → 仓位上限降到 2% ──
    it = tags.find("ST警示");
    if (it != tags.end() && it->second == "true") {
        m_blockedSymbols.insert(symbol);
        if (m_overrides.find(symbol) == m_overrides.end()) {
            m_overrides[symbol] = {
                config.stopLossPercent,
                config.maxPositionPercent
            };
        }
        config.maxPositionPercent = std::min(config.maxPositionPercent, 2.0);
        configChanged = true;
    }

    // ── 规则3: 高置信度负面政策 → 降总敞口 ──
    if (sentiment < -0.7 && confidence > 0.8 &&
        event.type() == engine::EventTypes::NEWS_POLICY) {
        config.maxTotalExposurePercent =
            std::min(config.maxTotalExposurePercent, 80.0);
        configChanged = true;
        INTERNAL_WARN_STREAM << "[EventRisk] 政策负面, 总敞口降至 "
                             << config.maxTotalExposurePercent << "%";
    }

    if (configChanged) {
        RiskManager::instance().setRiskConfig(config);
    }
}
```

## 五、错误处理策略

| 层级 | 异常类型 | 处理方式 |
|------|---------|---------|
| Python Adapter | 网络超时/API 限流 | 日志 + 跳过本轮 + 下一轮重试 |
| Python NLP Pipeline | HanLP 模型加载失败 | 日志 + 降级到纯词典模式 |
| Python NLP Pipeline | 单条文本分析异常 | 日志 + 跳过该条 + 继续处理下一条 |
| Python Publisher | EventBus 不可用 | 日志 + 事件丢弃 + 告警 |
| C++ EventDrivenFactor | 事件格式解析失败 | 日志 + 跳过该事件 |
| C++ EventRiskSubscriber | 事件无效/无标的 | 静默跳过 |
| C++ FactorEngine | EventDrivenFactor 计算异常 | try-catch, 返回空因子值 |
| C++ RiskManager | setRiskConfig 无效值 | 值域 clamp 保护 |

**全局原则**: 单条事件/单标的异常不影响其他事件和其他标的的处理。
