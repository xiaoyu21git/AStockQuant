cmake_minimum_required(VERSION 3.10)
project(JujinEventBusTest)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 添加可执行文件
add_executable(jujin_eventbus_test
    test_jujin_eventbus_simple.cpp
)

# 设置包含目录
target_include_directories(jujin_eventbus_test
    PRIVATE
        src/thirdParty/include
        src/engine/Event
)

# 设置编译选项
if(MSVC)
    target_compile_options(jujin_eventbus_test PRIVATE /W4 /EHsc)
else()
    target_compile_options(jujin_eventbus_test PRIVATE -Wall -Wextra -pedantic)
endif()
