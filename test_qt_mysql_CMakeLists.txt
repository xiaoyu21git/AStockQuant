cmake_minimum_required(VERSION 3.15)
project(test_qt_mysql VERSION 0.0.1)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 查找Qt6
find_package(Qt6 REQUIRED COMPONENTS Core Sql)

# 创建可执行文件
add_executable(test_qt_mysql_direct test_qt_mysql_direct.cpp)

# 链接Qt库
target_link_libraries(test_qt_mysql_direct PRIVATE Qt6::Core Qt6::Sql)

# 设置输出目录
set_target_properties(test_qt_mysql_direct PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin"
)
