cmake_minimum_required(VERSION 3.15)
project(test_data_cleaning VERSION 0.0.1)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 查找Qt6
find_package(Qt6 REQUIRED COMPONENTS Core)

# 包含项目头文件
include_directories(${CMAKE_SOURCE_DIR}/src/ui/bridge/include)

# 添加DataCleaningEngine源文件
set(DATA_CLEANING_SOURCES
    ${CMAKE_SOURCE_DIR}/src/ui/bridge/src/DataCleaningEngine.cpp
)

# 创建可执行文件
add_executable(test_data_cleaning 
    test_data_cleaning_complete.cpp
    ${DATA_CLEANING_SOURCES}
)

# 链接Qt库
target_link_libraries(test_data_cleaning PRIVATE Qt6::Core)

# 设置输出目录
set_target_properties(test_data_cleaning PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin"
)
